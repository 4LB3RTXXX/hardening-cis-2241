# ============================================================
# 04-ejecucion-y-pruebas.txt
# Ejecución del script de hardening y pruebas de las reglas CIS
# en R1-2241, SW1-2241 y FGT-CIS-LAB.
# Cada bloque indica en qué máquina o consola se ejecuta.
# ============================================================

# ------------------------------------------------------------
# A. En el Ubuntu auditor (zabbix-2241), desde la raíz del proyecto
# ------------------------------------------------------------
cd ~/hardening-cis-2241
source venv/bin/activate

# Auditoría de los tres equipos: genera informe SIN aplicar ninguna regla
python scripts/hardening.py --audit

# Auditoría de un solo equipo
python scripts/hardening.py --audit --equipos FGT-CIS-LAB

# Aplicar las reglas y generar informe póstumo
python scripts/hardening.py --apply
python scripts/hardening.py --apply --equipos FGT-CIS-LAB

# ------------------------------------------------------------
# B. En el Ubuntu auditor: pruebas de puertos
# ------------------------------------------------------------
# ANTES de aplicar: los puertos deben estar abiertos (succeeded)
date
nc -zv -w 3 10.22.41.1 23
nc -zv -w 3 10.22.41.2 23
nc -zv -w 3 10.22.41.2 80
nc -zv -w 3 10.22.41.3 23
nc -zv -w 3 10.22.41.3 80

# DESPUÉS de aplicar: deben fallar (Connection refused o timed out)
date
nc -zv -w 3 10.22.41.1 23
nc -zv -w 3 10.22.41.2 23
nc -zv -w 3 10.22.41.2 80
nc -zv -w 3 10.22.41.3 23
nc -zv -w 3 10.22.41.3 80

# Diferencias entre la configuración de antes y la de después
diff running-configs/R1-2241_antes.cfg running-configs/R1-2241_despues.cfg
diff running-configs/SW1-2241_antes.cfg running-configs/SW1-2241_despues.cfg
diff running-configs/FGT-CIS-LAB_antes.cfg running-configs/FGT-CIS-LAB_despues.cfg

# ------------------------------------------------------------
# C. En el Ubuntu auditor: entrar a R1-2241 por SSH
#    (las opciones -o son necesarias por los algoritmos antiguos del IOS)
# ------------------------------------------------------------
ssh -o KexAlgorithms=+diffie-hellman-group14-sha1 -o HostKeyAlgorithms=+ssh-rsa -o Ciphers=+aes128-cbc,aes256-cbc admin-local@10.22.41.1

# En la sesión SSH de R1-2241 (prompt R1-2241#):
show running-config | include enable secret|service password-encryption|no service pad|no ip source-route|service timestamps|no ip http
show running-config | section banner
show running-config | section line
show running-config interface FastEthernet0/0
exit

# ------------------------------------------------------------
# D. En el Ubuntu auditor: entrar a SW1-2241 por SSH
# ------------------------------------------------------------
ssh -o KexAlgorithms=+diffie-hellman-group14-sha1 -o HostKeyAlgorithms=+ssh-rsa -o Ciphers=+aes128-cbc,aes256-cbc admin-local@10.22.41.2

# En la sesión SSH de SW1-2241 (prompt SW1-2241#):
show running-config | include enable secret|service password-encryption|no service pad|service timestamps|no ip http
show running-config | section banner
show running-config | section line
exit

# ------------------------------------------------------------
# E. En el Ubuntu auditor: entrar a FGT-CIS-LAB por SSH
#    (debe aparecer el banner previo al inicio de sesión)
# ------------------------------------------------------------
ssh admin@10.22.41.3

# En la sesión SSH de FGT-CIS-LAB (prompt FGT-CIS-LAB #):
get system global | grep admintimeout
get system global | grep admin-lockout
get system global | grep pre-login-banner
show system password-policy
show system interface port1
exit
