# ============================================================
# 03-ubuntu-auditor-instalacion.txt
# Dónde se ejecuta: consola del Ubuntu auditor (zabbix-2241)
# Qué hace: instala Python, crea el entorno virtual e instala Netmiko.
# Requiere internet temporal: conectar el Ubuntu (ens3) a un nodo NAT
# de GNS3, instalar, y luego reconectarlo al switch SW1-2241.
# ============================================================

# --- 1. Con el Ubuntu conectado a NAT1 (internet temporal) ---
sudo netplan apply
ip -br a
ping -c 3 8.8.8.8
ping -c 3 pypi.org

# --- 2. Paquetes y entorno virtual ---
sudo apt update
sudo apt install -y python3-pip python3-venv git
mkdir -p ~/hardening-cis-2241
cd ~/hardening-cis-2241
python3 -m venv venv
source venv/bin/activate

# --- 3. Librerías (Paramiko 3.5.1 es obligatorio) ---
# Con Paramiko 5.x falla con "no acceptable kex algorithm", porque el IOS
# emulado solo ofrece intercambio de claves SHA-1.
pip install netmiko==4.8.0
pip install "paramiko==3.5.1"
python -c "import paramiko; print(paramiko.__version__)"

# --- 4. Copiar la carpeta scripts del repositorio al Ubuntu ---
# (por ejemplo con scp desde PowerShell de la PC de administración)
# scp -r scripts albert@10.22.41.50:~/hardening-cis-2241/

# --- 5. Reconectar el Ubuntu al switch SW1-2241 en GNS3 y verificar ---
sudo netplan apply
ip -br a
ping -c 3 10.22.41.1
ping -c 3 10.22.41.2
