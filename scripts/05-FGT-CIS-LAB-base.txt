# ============================================================
# 05-FGT-CIS-LAB-base.txt
# Dónde se ejecuta: consola de FGT-CIS-LAB (FortiGate 7.0.9, GNS3)
# Qué hace: configuración base del firewall, dejando a propósito
#           opciones inseguras abiertas para que el script de
#           hardening las detecte (antes) y las corrija (después).
# Conexión: port1 del FortiGate al switch SW1-2241.
# ============================================================

# --- 1. Configuración base (estado ANTES del hardening) ---
config system global
    set hostname "FGT-CIS-LAB"
    set admintimeout 30
    set admin-lockout-threshold 10
    set admin-lockout-duration 60
end

config system password-policy
    set status disable
end

config system interface
    edit "port1"
        set mode static
        set ip 10.22.41.3 255.255.255.0
        set allowaccess ping https http ssh telnet
    next
end

# --- 2. Verificación ---
# get system status
# execute ping 10.22.41.1

# --- 3. Revertir al estado inseguro (para repetir la demostración) ---
config system global
    set admintimeout 30
    set admin-lockout-threshold 10
    set admin-lockout-duration 60
    set pre-login-banner disable
end

config system password-policy
    set status disable
end

config system interface
    edit "port1"
        set allowaccess ping https http ssh telnet
    next
end
